<script>
(function () {
  var files = [
    "app/(auth)/entities/[entityId]/users/page.tsx",
    "app/(auth)/entities/[entityId]/users/_components/EntityHeader.tsx",
    "app/(auth)/entities/[entityId]/users/_components/UsersPageTable.tsx",
    "app/api/entities/[entityId]/route.ts",
    "app/api/users/route.ts",
    "service/entities/entity.service.ts",
    "service/entities/entity.types.ts",
    "service/entities/schema/entity.schema.ts",
    "service/users/schema/user.schema.ts",
    "service/users/user.mapper.ts",
    "service/users/user.repository.ts",
    "service/users/user.service.ts",
    "service/users/user.types.ts"
  ];

  var root = document.getElementById("mock-next-source-browser");
  var currentCode = "";

  if (!root || root.dataset.ready === "true") {
    return;
  }

  root.dataset.ready = "true";
  root.innerHTML =
    '<aside class="mock-next-sidebar">' +
      '<div class="mock-next-sidebar-header">' +
        '<div class="mock-next-title">mock-next</div>' +
        '<input class="mock-next-search" type="search" placeholder="过滤文件" aria-label="过滤文件">' +
      '</div>' +
      '<nav class="mock-next-tree" aria-label="mock-next 文件目录"></nav>' +
    '</aside>' +
    '<section class="mock-next-main">' +
      '<div class="mock-next-toolbar">' +
        '<div class="mock-next-path"></div>' +
        '<div class="mock-next-actions">' +
          '<button class="mock-next-copy" type="button">复制源码</button>' +
        '</div>' +
      '</div>' +
      '<div class="mock-next-code-wrap">' +
        '<pre class="mock-next-code" aria-live="polite"></pre>' +
      '</div>' +
    '</section>';

  var searchInput = root.querySelector(".mock-next-search");
  var treeElement = root.querySelector(".mock-next-tree");
  var pathElement = root.querySelector(".mock-next-path");
  var codeElement = root.querySelector(".mock-next-code");
  var copyButton = root.querySelector(".mock-next-copy");

  function escapeHtml(value) {
    return value
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&#039;");
  }

  function toFileUrl(filePath) {
    return ["mock-next"].concat(filePath.split("/"))
      .map(encodeURIComponent)
      .join("/");
  }

  function createTree(paths) {
    var tree = { dirs: {}, files: [] };

    paths.forEach(function (filePath) {
      var parts = filePath.split("/");
      var cursor = tree;

      parts.forEach(function (part, index) {
        if (index === parts.length - 1) {
          cursor.files.push({ name: part, path: filePath });
          return;
        }

        if (!cursor.dirs[part]) {
          cursor.dirs[part] = { dirs: {}, files: [] };
        }

        cursor = cursor.dirs[part];
      });
    });

    return tree;
  }

  function renderTreeNode(node) {
    var dirNames = Object.keys(node.dirs).sort();
    var fileItems = node.files.slice().sort(function (a, b) {
      return a.name.localeCompare(b.name);
    });
    var html = "<ul>";

    dirNames.forEach(function (dirName) {
      html +=
        "<li>" +
          '<div class="mock-next-folder">' + escapeHtml(dirName) + "</div>" +
          renderTreeNode(node.dirs[dirName]) +
        "</li>";
    });

    fileItems.forEach(function (file) {
      html +=
        "<li>" +
          '<button class="mock-next-file" type="button" data-file="' +
          escapeHtml(file.path) +
          '">' +
          escapeHtml(file.name) +
          "</button>" +
        "</li>";
    });

    html += "</ul>";
    return html;
  }

  function renderTree(filterText) {
    var normalizedFilter = filterText.trim().toLowerCase();
    var visibleFiles = normalizedFilter
      ? files.filter(function (filePath) {
          return filePath.toLowerCase().includes(normalizedFilter);
        })
      : files;

    if (visibleFiles.length === 0) {
      treeElement.innerHTML = '<div class="mock-next-empty">没有匹配的文件</div>';
      return;
    }

    treeElement.innerHTML = renderTreeNode(createTree(visibleFiles));
  }

  function renderCode(code, language) {
    var lines = code.split("\n");

    codeElement.innerHTML = lines.map(function (line, index) {
      var highlightedLine = line ? highlightLine(line, language) : " ";

      return (
        '<span class="mock-next-line">' +
          '<span class="mock-next-line-number">' + (index + 1) + "</span>" +
          '<span class="mock-next-line-code">' +
            highlightedLine +
          "</span>" +
        "</span>"
      );
    }).join("");
  }

  function setActiveFile(filePath) {
    root.querySelectorAll(".mock-next-file").forEach(function (button) {
      button.classList.toggle("is-active", button.dataset.file === filePath);
    });
  }

  function showMessage(message) {
    codeElement.innerHTML = '<div class="mock-next-message">' + escapeHtml(message) + "</div>";
  }

  function getLanguage(filePath) {
    if (filePath.endsWith(".md")) {
      return "markdown";
    }

    if (filePath.endsWith(".tsx") || filePath.endsWith(".ts")) {
      return "javascript";
    }

    if (filePath.endsWith(".json")) {
      return "json";
    }

    return "javascript";
  }

  function highlightLine(line, language) {
    if (!window.Prism || !window.Prism.languages) {
      return escapeHtml(line);
    }

    var grammar = window.Prism.languages[language] || window.Prism.languages.javascript;

    if (!grammar) {
      return escapeHtml(line);
    }

    return window.Prism.highlight(line, grammar, language);
  }

  function loadFile(filePath) {
    pathElement.textContent = "mock-next/" + filePath;
    setActiveFile(filePath);
    showMessage("正在加载 " + filePath + " ...");

    fetch(toFileUrl(filePath), { cache: "no-store" })
      .then(function (response) {
        if (!response.ok) {
          throw new Error("HTTP " + response.status);
        }

        return response.text();
      })
      .then(function (code) {
        currentCode = code;
        renderCode(code, getLanguage(filePath));
      })
      .catch(function (error) {
        currentCode = "";
        showMessage("源码加载失败：" + error.message);
      });
  }

  treeElement.addEventListener("click", function (event) {
    var button = event.target.closest(".mock-next-file");

    if (!button) {
      return;
    }

    loadFile(button.dataset.file);
  });

  searchInput.addEventListener("input", function () {
    renderTree(searchInput.value);
    setActiveFile(pathElement.textContent.replace(/^mock-next\//, ""));
  });

  copyButton.addEventListener("click", function () {
    if (!currentCode || !navigator.clipboard) {
      return;
    }

    navigator.clipboard.writeText(currentCode).then(function () {
      var oldText = copyButton.textContent;
      copyButton.textContent = "已复制";

      window.setTimeout(function () {
        copyButton.textContent = oldText;
      }, 1200);
    });
  });

  renderTree("");
  loadFile("app/(auth)/entities/[entityId]/users/page.tsx");
}());
</script>
